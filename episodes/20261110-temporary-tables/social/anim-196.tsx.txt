import React, {createContext, useContext} from 'react';
import {AbsoluteFill, Easing, interpolate, useCurrentFrame, useVideoConfig} from 'remotion';
import {font} from '../font';

// Postgres.FM #196 social animation, "app window on a wallpaper" style. Silent, portrait 1080x1350.
// Rules: one idea per slide; every number labels a visual drawn to true scale; numbers appear
// only as final measured values (no count-ups). Fact: the ~23x gap at 0 idle sessions is catalog
// rows + file create/unlink work; the shared invalidation queue is a SECOND cost that appears
// with many idle sessions. Numbers: the fact-checked set (brief throughput and cliff figures, ring card).

const MONO = 'JetBrains Mono, Menlo, monospace';
const SANS = 'Inter, sans-serif';
const K = {bg: '#0a0a0a', line: 'rgba(255,255,255,0.09)', dim: 'rgba(255,255,255,0.62)', faint: 'rgba(255,255,255,0.42)',
  green: '#3DDC84', red: '#FF6B5B', yellow: '#FFD23F', blue: '#7CC4FF', orange: '#FF8A3D', white: '#F5F5F5', track: 'rgba(255,255,255,0.06)'};
const clamp = {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'} as const;
const EASE = Easing.bezier(0.22, 1, 0.36, 1);
const fmt = (n: number) => Math.round(n).toLocaleString('en-US');

// A time override: the loop poster renders slide 1 frozen at t = 0.
const TCtx = createContext<number | null>(null);
const useT = () => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const over = useContext(TCtx);
  font();
  const t = over ?? frame / fps;
  const e = (a: number, d = 0.7) => interpolate(t, [a, a + d], [0, 1], {...clamp, easing: EASE});
  return {t, e};
};

const Fade: React.FC<{at: number; d?: number; y?: number; style?: React.CSSProperties; children: React.ReactNode}> = ({at, d = 0.7, y = 18, style, children}) => {
  const {e} = useT();
  const p = e(at, d);
  return <div style={{opacity: p, transform: `translateY(${(1 - p) * y}px)`, ...style}}>{children}</div>;
};
const Eyebrow: React.FC<{at: number; color: string; text: string}> = ({at, color, text}) => (
  <Fade at={at} style={{display: 'flex', alignItems: 'center', gap: 14, fontFamily: MONO, fontSize: 26, letterSpacing: '0.12em', color: K.dim}}>
    <span style={{width: 12, height: 12, borderRadius: 6, background: color, boxShadow: `0 0 14px ${color}`}} />
    {text}
  </Fade>
);
const Head: React.FC<{at: number; size: number; children: React.ReactNode; style?: React.CSSProperties}> = ({at, size, children, style}) => (
  <Fade at={at} y={26} style={{fontFamily: SANS, fontWeight: 600, fontSize: size, letterSpacing: '-0.035em', lineHeight: 1.06, color: K.white, marginTop: 20, ...style}}>
    {children}
  </Fade>
);
const Tiny: React.FC<{at: number; children: React.ReactNode; style?: React.CSSProperties}> = ({at, children, style}) => (
  <Fade at={at} style={{fontFamily: MONO, fontSize: 26, color: K.faint, letterSpacing: '0.04em', ...style}}>{children}</Fade>
);

// Typed code line (the bar's label) with a block cursor while typing.
const Typed: React.FC<{at: number; text: string; color: string; size: number; cps?: number}> = ({at, text, color, size, cps = 40}) => {
  const {t} = useT();
  const n = Math.max(0, Math.floor((t - at) * cps));
  const done = n >= text.length;
  const blink = Math.floor(t * 2.2) % 2 === 0;
  return (
    <div style={{fontFamily: MONO, fontSize: size, lineHeight: 1.35, color: K.white, whiteSpace: 'pre', height: size * 1.35}}>
      <span style={{color}}>=# </span>{t >= at ? text.slice(0, n) : ''}
      <span style={{display: 'inline-block', width: size * 0.58, height: size * 1.1, verticalAlign: 'text-bottom',
        background: t >= at && (!done || t - at - text.length / cps < 0.5) && blink ? K.white : 'transparent'}} />
    </div>
  );
};

// ---------- slide: the race (bars to true scale) ----------
const Race: React.FC<{w: number}> = ({w}) => {
  const {t, e} = useT();
  const r0 = 1.5, rd = 1.8, punchAt = 3.4;
  const MAX = 88531;
  const k = interpolate(t, [r0, r0 + rd], [0, 1], {...clamp, easing: Easing.out(Easing.cubic)});
  const BH = 84;
  // A bar's final value fades in when the bar itself reaches it (same TPS rate for both bars).
  const doneAt = (val: number) => {
    const f = val / MAX; // invert out-cubic: k = 1-(1-x)^3
    return r0 + rd * (1 - Math.cbrt(1 - f));
  };
  const punch = e(punchAt, 0.7);
  const bar = (val: number, color: string, inside: boolean, extra?: React.ReactNode) => {
    const bw = (w * Math.min(val, MAX * k)) / MAX;
    const lab = e(doneAt(val), 0.4);
    return (
      <div style={{position: 'relative', height: BH, background: K.track, borderRadius: 10, marginTop: 16}}>
        <div style={{position: 'absolute', left: 0, top: 0, height: BH, width: Math.max(bw, k > 0 ? 8 : 0), background: color, borderRadius: 10,
          boxShadow: `0 0 26px ${color}55`}} />
        <div style={{position: 'absolute', top: 0, height: BH, display: 'flex', alignItems: 'center', fontFamily: SANS, fontWeight: 600, fontSize: 42,
          letterSpacing: '-0.02em', opacity: lab, whiteSpace: 'nowrap',
          ...(inside ? {right: 22, color: '#06210F'} : {left: Math.max(bw, 8) + 18, color})}}>
          {fmt(val)} <span style={{fontSize: 28, fontFamily: MONO, marginLeft: 10, opacity: 0.85}}>TPS</span>
        </div>
        {extra}
      </div>
    );
  };
  return (
    <div>
      <Typed at={0.2} text="create temp table t …" color={K.red} size={42} />
      {bar(3931, K.red, false,
        <div style={{position: 'absolute', right: 22, top: 0, height: BH, display: 'flex', alignItems: 'center', gap: 12, opacity: punch,
          transform: `translateX(${(1 - punch) * 30}px)`}}>
          <span style={{fontFamily: SANS, fontWeight: 700, fontSize: 62, letterSpacing: '-0.04em', color: K.red}}>~23×</span>
          <span style={{fontFamily: SANS, fontWeight: 600, fontSize: 40, color: K.red}}>slower</span>
        </div>)}
      <div style={{height: 52}} />
      <Typed at={0.85} text="with t as (…) select …" color={K.green} size={42} />
      {bar(88531, K.green, true)}
      <Fade at={4.1} style={{fontFamily: MONO, fontSize: 30, color: K.dim, marginTop: 40}}>cost: catalog rows + files, every transaction</Fade>
    </div>
  );
};

// ---------- slide: the queue (a second cost), ~114 messages per transaction ----------
// 108 ticks x ~38 slots = 4,096; one create + drop (~114 messages) = 3 ticks; 36 transactions fill the ring.
const qT = (at: number) => ({d0: at + 2.2, d1: at + 4.0, d2: at + 7.6});
const txCount = (t: number, at: number) => {
  const {d0, d1, d2} = qT(at);
  if (t < d0) return 0;
  if (t < d1) return 1;
  return 1 + Math.floor(interpolate(t, [d1, d2], [0, 35], {...clamp, easing: Easing.in(Easing.quad)}) + 1e-6);
};
const Ring: React.FC<{at: number; w: number; h: number}> = ({at, w, h}) => {
  const {t, e} = useT();
  const N = 108, PER = 3;
  const {d0} = qT(at);
  const cx = w / 2, cy = h * 0.6, rx = Math.min(w * 0.42, 350), ry = rx * 0.5, PH = rx * 0.17;
  const tx = txCount(t, at);
  const lit = tx * PER;
  const full = tx >= 36;
  const pillars = Array.from({length: N}, (_, i) => {
    const a = (i / N) * Math.PI * 2 - Math.PI / 2;
    return {i, x: cx + Math.cos(a) * rx, y: cy + Math.sin(a) * ry};
  }).sort((p, q) => p.y - q.y);
  const FALL = 90;
  const fall = interpolate(t, [d0 - 1.0, d0], [0, 1], {...clamp, easing: Easing.in(Easing.cubic)});
  const a0 = -Math.PI / 2 + (1.5 / N) * Math.PI * 2;
  const tx0 = cx + Math.cos(a0) * rx, ty0 = cy + Math.sin(a0) * ry - PH;
  return (
    <svg width={w} height={h} style={{display: 'block', opacity: e(at + 0.5, 0.9), overflow: 'visible'}}>
      <defs>
        <filter id="glow" filterUnits="userSpaceOnUse" x={-100} y={-300} width={w + 200} height={h + 500}><feGaussianBlur stdDeviation="4" result="b" /><feMerge><feMergeNode in="b" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
      </defs>
      <ellipse cx={cx} cy={cy} rx={rx} ry={ry} fill="none" stroke="rgba(255,255,255,0.10)" strokeWidth={1.5} strokeDasharray="3 7" />
      {pillars.map((p) => {
        const isLit = p.i < lit;
        const newest = p.i >= lit - PER && p.i < lit;
        const col = isLit ? (newest && !full ? K.white : full ? K.red : K.orange) : 'rgba(255,255,255,0.22)';
        return <line key={p.i} x1={p.x} y1={p.y} x2={p.x} y2={p.y - PH} stroke={col} strokeWidth={isLit ? 6 : 4} strokeLinecap="round"
          filter={isLit ? 'url(#glow)' : undefined} />;
      })}
      {t < d0 + 0.2 && t > d0 - 1.2 && (
        <g opacity={interpolate(t, [d0 - 1.2, d0 - 0.9, d0, d0 + 0.2], [0, 1, 1, 0], clamp)}>
          {[-1, 0, 1].map((j) => <line key={j} x1={tx0 + j * 16} x2={tx0 + j * 16} y1={ty0 - FALL + fall * FALL} y2={ty0 - FALL + fall * FALL + PH}
            stroke={K.white} strokeWidth={6} strokeLinecap="round" filter="url(#glow)" />)}
        </g>
      )}
      <text x={cx} y={cy + 14} textAnchor="middle" fill={K.white} fontFamily={SANS} fontWeight={600} fontSize={96} letterSpacing="-3">4,096</text>
      <text x={cx} y={cy + 58} textAnchor="middle" fill={K.dim} fontFamily={MONO} fontSize={28} letterSpacing="2">SLOTS</text>
    </svg>
  );
};
const RingLegend: React.FC<{at: number}> = ({at}) => {
  const {t, e} = useT();
  const {d0, d1} = qT(at);
  const tx = txCount(t, at);
  const chip = (c: string) => <span style={{display: 'inline-block', width: 8, height: 38, borderRadius: 4, background: c, marginRight: 7, boxShadow: `0 0 10px ${c}`}} />;
  return (
    <div style={{fontFamily: SANS, fontWeight: 600, fontSize: 42, color: K.white, letterSpacing: '-0.02em'}}>
      <div style={{display: 'flex', alignItems: 'center', gap: 20, opacity: e(d0 - 1.0)}}>
        <span>{chip(K.white)}{chip(K.white)}{chip(K.white)}</span>
        <span>1 create + drop ≈ <span style={{color: K.orange}}>114 messages</span></span>
      </div>
      <div style={{display: 'flex', alignItems: 'center', gap: 20, marginTop: 22, opacity: e(d1 - 0.2)}}>
        <span style={{fontFamily: MONO, fontSize: 38, color: tx >= 36 ? K.red : K.orange, minWidth: 140}}>× {tx}</span>
        <span>{tx >= 36 ? <>= full: <span style={{color: K.red}}>laggards reset</span></> : 'transactions'}</span>
      </div>
    </div>
  );
};

// ---------- slide: every session reads every message ----------
const Sessions: React.FC<{at: number; w: number; h: number}> = ({at, w, h}) => {
  const {t, e} = useT();
  const G = 14, s = 30;
  const ox = w / 2, oy = h / 2 - G * s * 0.5 + 10;
  const t0 = at + 1.6;
  const tiles: React.ReactNode[] = [];
  for (let r = 0; r < G; r++) for (let c = 0; c < G; c++) {
    const x = ox + (c - r) * s, y = oy + (c + r) * s * 0.5;
    const ph = ((t - t0) * 0.55 - ((r + c) / (2 * G)) * 0.8) % 1;
    const k = t > t0 && ph > 0 && ph < 0.32 ? 1 - ph / 0.32 : 0;
    const col = k > 0 ? K.orange : 'rgba(124,196,255,0.60)';
    tiles.push(<path key={`${r}-${c}`} d={`M${x},${y - s * 0.43} L${x + s * 0.86},${y} L${x},${y + s * 0.43} L${x - s * 0.86},${y} Z`}
      fill={k > 0 ? col : 'none'} fillOpacity={0.25 + 0.6 * k} stroke={col} strokeWidth={1.6} opacity={0.75 + 0.25 * k} />);
  }
  // one tile called out
  const cx = ox + (G - 1) * s, cy = oy + (G - 1) * s * 0.5; // right corner tile
  return (
    <svg width={w} height={h} style={{display: 'block', opacity: e(at + 0.5, 0.9), overflow: 'visible'}}>
      <defs>
        <filter id="glow3" filterUnits="userSpaceOnUse" x={-100} y={-100} width={w + 200} height={h + 200}><feGaussianBlur stdDeviation="4" result="b" /><feMerge><feMergeNode in="b" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
      </defs>
      <g filter="url(#glow3)">{tiles}</g>
      <g opacity={e(at + 1.1)}>
        <path d={`M${cx},${cy - s * 0.43} L${cx + s * 0.86},${cy} L${cx},${cy + s * 0.43} L${cx - s * 0.86},${cy} Z`} fill="none" stroke={K.white} strokeWidth={3} />
        <line x1={cx - 4} y1={cy + s * 0.5} x2={cx - 24} y2={h - 50} stroke={K.white} strokeWidth={2} />
        <text x={w} y={h - 14} textAnchor="end" fill={K.white} fontFamily={MONO} fontSize={28}>each tile = one session</text>
      </g>
    </svg>
  );
};

// ---------- slide: the cliff ----------
const Cliff: React.FC<{at: number; w: number; h: number}> = ({at, w, h}) => {
  const {e} = useT();
  const pts = [[0, 502], [1000, 502], [3000, 274], [5000, 127]];
  const L = 92, R = w - 30, T = 80, Bt = h - 100;
  const X = (v: number) => L + ((R - L) * v) / 5000;
  const Y = (v: number) => Bt - ((Bt - T) * v) / 560;
  const D0 = at + 1.2, DRAW = 3.2;
  const draw = interpolate(useT().t, [D0, D0 + DRAW], [0, 1], clamp); // linear so labels can sync to the pen
  // path length per segment, for syncing labels to the moment the line reaches each point
  const seg = pts.slice(1).map((p, i) => Math.hypot(X(p[0]) - X(pts[i][0]), Y(p[1]) - Y(pts[i][1])));
  const total = seg.reduce((a, b) => a + b, 0);
  const reach = [0, ...seg.map((_, i) => seg.slice(0, i + 1).reduce((a, b) => a + b, 0) / total)];
  const d = pts.map((p, i) => `${i ? 'L' : 'M'}${X(p[0])},${Y(p[1])}`).join(' ');
  const cols = [K.green, K.green, K.yellow, K.red];
  // label placement off the line: above the flat start, upper right of 274, below left of 127
  const lab = [
    {x: (X(0) + X(1000)) / 2, y: Y(502) - 30, a: 'middle'},
    null,
    {x: X(3000) + 22, y: Y(274) - 26, a: 'start'},
    {x: X(5000) - 18, y: Y(127) + 66, a: 'end'},
  ];
  return (
    <svg width={w} height={h} style={{display: 'block', overflow: 'visible'}}>
      <defs>
        <linearGradient id="lg" x1="0" x2="1" y1="0" y2="0"><stop offset="0.2" stopColor={K.green} /><stop offset="0.6" stopColor={K.yellow} /><stop offset="1" stopColor={K.red} /></linearGradient>
        <filter id="glow2" filterUnits="userSpaceOnUse" x={-100} y={-100} width={w + 200} height={h + 200}><feGaussianBlur stdDeviation="5" result="b" /><feMerge><feMergeNode in="b" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
      </defs>
      <g opacity={e(at + 0.6)}>
        {[0, 250].map((v) => (
          <g key={v}>
            <line x1={L} x2={R} y1={Y(v)} y2={Y(v)} stroke="rgba(255,255,255,0.12)" strokeDasharray="2 8" strokeWidth={1.5} />
            <text x={L - 16} y={Y(v) + 9} textAnchor="end" fill={K.dim} fontFamily={MONO} fontSize={26}>{v}</text>
          </g>
        ))}
        <line x1={L} x2={R} y1={Y(500)} y2={Y(500)} stroke="rgba(255,255,255,0.45)" strokeDasharray="10 8" strokeWidth={2} />
        <text x={L - 16} y={Y(500) + 9} textAnchor="end" fill={K.dim} fontFamily={MONO} fontSize={26}>500</text>
        <text x={R} y={Y(500) - 16} textAnchor="end" fill={K.dim} fontFamily={MONO} fontSize={28}>target: 500 TPS</text>
        <text x={L - 16} y={T - 34} textAnchor="end" fill={K.dim} fontFamily={MONO} fontSize={26}>TPS</text>
        {pts.map((p, i) => (
          <text key={i} x={X(p[0])} y={Bt + 48} textAnchor={i === 0 ? 'start' : i === 3 ? 'end' : 'middle'} fill={K.dim} fontFamily={MONO} fontSize={30}>{fmt(p[0])}</text>
        ))}
        <text x={(L + R) / 2} y={Bt + 92} textAnchor="middle" fill={K.dim} fontFamily={MONO} fontSize={28} letterSpacing="1">idle sessions</text>
      </g>
      <path d={d} pathLength={1} fill="none" stroke="url(#lg)" strokeWidth={6} strokeLinejoin="round" strokeDasharray="1 1" strokeDashoffset={1 - draw} filter="url(#glow2)" />
      {pts.map((p, i) => {
        const appear = e(D0 + DRAW * reach[i] - 0.05, 0.4);
        const l = lab[i];
        return (
          <g key={i} opacity={appear}>
            <circle cx={X(p[0])} cy={Y(p[1])} r={11} fill={K.bg} stroke={cols[i]} strokeWidth={4} />
            {l && <text x={l.x} y={l.y} textAnchor={l.a as 'start'} fill={cols[i]} fontFamily={SANS} fontWeight={600} fontSize={54} letterSpacing="-1.5">
              {fmt(p[1])}<tspan fontFamily={MONO} fontSize={28} fontWeight={400} dx={8}>TPS</tspan>
            </text>}
          </g>
        );
      })}
    </svg>
  );
};

// ---------- end card joke: "$ listen postgres.fm" -> backspace "listen" -> "watch" ----------
const Joke: React.FC<{at: number}> = ({at}) => {
  const {t} = useT();
  const a = t - at;
  const rest = ' postgres.fm';
  let pre = '', post = '';
  if (a < 0) pre = '';
  else if (a < 0.8) pre = ('listen' + rest).slice(0, Math.ceil((a / 0.8) * 18));
  else if (a < 1.3) pre = 'listen' + rest;
  else if (a < 1.7) { pre = 'listen'.slice(0, 6 - Math.floor(((a - 1.3) / 0.4) * 6)); post = rest; }
  else if (a < 2.1) { pre = 'watch'.slice(0, Math.floor(((a - 1.7) / 0.4) * 5)); post = rest; }
  else pre = 'watch' + rest;
  const editing = a >= 1.3 && a < 2.1;
  const on = editing || Math.floor(t * 2.2) % 2 === 0;
  return (
    <span>
      <span style={{color: K.green}}>$ </span>{pre}
      <span style={{display: 'inline-block', width: 26, height: 48, verticalAlign: 'text-bottom', background: on ? K.white : 'transparent', opacity: editing ? 0.85 : 1}} />
      {post}
    </span>
  );
};

// ---------- wallpaper ----------
const Wallpaper: React.FC<{W: number; H: number}> = ({W, H}) => {
  const frame = useCurrentFrame();
  const {durationInFrames} = useVideoConfig();
  // phase loops over the composition, so the last frame matches the first
  const t = (frame / durationInFrames) * Math.PI * 2;
  const blob = (x: number, y: number, r: number, c: string, ph: number) => (
    <div style={{position: 'absolute', left: x + Math.sin(t + ph) * 60 - r, top: y + Math.cos(t + ph) * 40 - r,
      width: 2 * r, height: 2 * r, borderRadius: '50%', background: c, filter: 'blur(90px)'}} />
  );
  const wave = (k: number, c: string, o: number) => {
    const pts: string[] = [];
    for (let x = -50; x <= W + 50; x += 30) {
      const y = H * (0.42 + k * 0.17) + Math.sin(x / (W * 0.18) + t + k * 1.7) * H * 0.07 + Math.sin(x / (W * 0.07) + k) * H * 0.015;
      pts.push(`${x},${y}`);
    }
    return <polygon points={`-50,${H + 50} ${pts.join(' ')} ${W + 50},${H + 50}`} fill={c} opacity={o} />;
  };
  return (
    <AbsoluteFill style={{background: 'linear-gradient(150deg, #120B33 0%, #1B1150 45%, #0B1C4A 100%)', overflow: 'hidden'}}>
      {blob(W * 0.15, H * 0.2, W * 0.35, '#5B2BB5', 0)}
      {blob(W * 0.85, H * 0.3, W * 0.32, '#1F56D6', 2)}
      {blob(W * 0.6, H * 0.95, W * 0.4, '#8A2E9E', 4)}
      <svg width={W} height={H} style={{position: 'absolute', inset: 0, filter: 'blur(14px)'}}>
        {wave(0, '#3A2A9A', 0.35)}
        {wave(1, '#5A2FA8', 0.32)}
        {wave(2, '#1C3FA0', 0.38)}
      </svg>
    </AbsoluteFill>
  );
};

// Schedules: section start times (s); content of a section starts 0.9 s before, while it scrolls in.
export const CUTS = {
  full: {ids: ['bars', 'queue', 'sessions', 'cliff', 'donut', 'fix', 'end'] as SlideId[], S: [0, 8.0, 18.2, 24.8, 31.6, 37.4, 42.4], dur: 49.0, joke: 1.5},
  x: {ids: ['bars', 'cliff', 'donut', 'fix', 'end'] as SlideId[], S: [0, 7.2, 13.6, 19.0, 22.2], dur: 25.0, joke: 0.6},
};
const LOOP = 0.8; // last LOOP seconds crossfade back to the opening frame

export const Win196: React.FC<{fmt: Fmt; cut?: keyof typeof CUTS}> = ({cut = 'full'}) => {
  const frame = useCurrentFrame();
  const {fps, width: W, height: H} = useVideoConfig();
  const t = frame / fps;
  const C = CUTS[cut];
  const e = (a: number, d = 0.7) => interpolate(t, [a, a + d], [0, 1], {...clamp, easing: EASE});
  font();
  const win = {x: 54, y: 70, w: W - 108, h: H - 140};
  const BAR = 50;
  const cw = win.w, ch = win.h - BAR;
  const pad = 60;
  const inner = cw - 2 * pad;
  const scroll = C.S.slice(1).reduce((acc, b) => acc + e(b - 0.45, 1.0), 0) * ch;
  const loop = interpolate(t, [C.dur - LOOP, C.dur - 1 / fps], [0, 1], {...clamp, easing: Easing.inOut(Easing.cubic)});
  const sec = (i: number, top: number, children: React.ReactNode) => (
    <div key={i} style={{position: 'absolute', left: 0, top, width: cw, height: ch, padding: `56px ${pad}px 52px`, boxSizing: 'border-box',
      display: 'flex', flexDirection: 'column', justifyContent: 'space-between'}}>{children}</div>
  );
  return (
    <AbsoluteFill>
      <Wallpaper W={W} H={H} />
      <div style={{position: 'absolute', left: win.x, top: win.y, width: win.w, height: win.h, borderRadius: 16, overflow: 'hidden',
        background: K.bg, border: '1px solid rgba(255,255,255,0.12)', boxShadow: '0 40px 120px rgba(0,0,0,0.55), 0 0 0 1px rgba(0,0,0,0.5)'}}>
        <div style={{height: BAR, display: 'flex', alignItems: 'center', padding: '0 20px', borderBottom: `1px solid ${K.line}`, position: 'relative', background: '#111113'}}>
          {['#FF5F57', '#FEBC2E', '#28C840'].map((c) => <span key={c} style={{width: 14, height: 14, borderRadius: 7, background: c, marginRight: 9}} />)}
          <div style={{position: 'absolute', left: 0, right: 0, textAlign: 'center', fontFamily: MONO, fontSize: 20, color: K.faint, letterSpacing: '0.04em'}}>
            postgres.fm — episode 196
          </div>
        </div>
        <div style={{position: 'relative', width: cw, height: ch, overflow: 'hidden'}}>
          <div style={{position: 'absolute', left: 0, top: -scroll, width: cw, opacity: 1 - loop}}>
            {C.ids.map((id, i) => sec(i, i * ch, SLIDES[id]({B: i ? C.S[i] - 0.9 : 0, inner, jokeAt: C.joke})))}
          </div>
          {loop > 0 && (
            <TCtx.Provider value={0}>
              <div style={{position: 'absolute', inset: 0, opacity: loop}}>
                {sec(99, 0, SLIDES.bars({B: 0, inner, jokeAt: 0}))}
              </div>
            </TCtx.Provider>
          )}
        </div>
      </div>
    </AbsoluteFill>
  );
};
type Fmt = 'portrait' | 'landscape';
export const DUR = CUTS.full.dur;
