import React from 'react';

// Postgres.FM #196 social animation, "app window on a wallpaper" style. Silent.
// Rule: no number stands alone; every number labels a visual drawn to true scale.
// Numbers: the fact-checked set (brief throughput and cliff figures, ring card).

const MONO = 'JetBrains Mono, Menlo, monospace';
const SANS = 'Inter, sans-serif';
const K = {bg: '#0a0a0a', line: 'rgba(255,255,255,0.09)', dim: 'rgba(255,255,255,0.62)', faint: 'rgba(255,255,255,0.40)',
  green: '#3DDC84', red: '#FF6B5B', yellow: '#FFD23F', blue: '#7CC4FF', orange: '#FF8A3D', white: '#F5F5F5', track: 'rgba(255,255,255,0.06)'};
const clamp = {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'} as const;
const EASE = Easing.bezier(0.22, 1, 0.36, 1);
const fmt = (n: number) => Math.round(n).toLocaleString('en-US');

// Section start times (s); the page scrolls around each boundary.
const S = [0, 8.8, 15.0, 21.0, 24.2];
export const DUR = 28.8;
// Content of a section starts while it scrolls in.
const B = S.map((v, i) => (i ? v - 0.9 : v));

type Fmt = 'portrait' | 'landscape';
const useT = () => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  font();
  const t = frame / fps;
  const e = (a: number, d = 0.7) => interpolate(t, [a, a + d], [0, 1], {...clamp, easing: EASE});
  return {t, e};
};

const Fade: React.FC<{at: number; d?: number; y?: number; style?: React.CSSProperties; children: React.ReactNode}> = ({at, d = 0.7, y = 18, style, children}) => {
  const {e} = useT();
  const p = e(at, d);
  return <div style={{opacity: p, transform: `translateY(${(1 - p) * y}px)`, ...style}}>{children}</div>;
};

// Typed code line with a block cursor while typing.
const Typed: React.FC<{at: number; text: string; color: string; size: number}> = ({at, text, color, size}) => {
  const {t} = useT();
  const CPS = 42;
  const n = Math.max(0, Math.floor((t - at) * CPS));
  const done = n >= text.length;
  const blink = Math.floor(t * 2.2) % 2 === 0;
  if (t < at) return <div style={{height: size * 1.4}} />;
  return (
    <div style={{fontFamily: MONO, fontSize: size, lineHeight: 1.4, color: K.white, whiteSpace: 'pre'}}>
      <span style={{color}}>=# </span>{text.slice(0, n)}
      <span style={{display: 'inline-block', width: size * 0.58, height: size * 1.1, verticalAlign: 'text-bottom',
        background: !done || t - at - text.length / CPS < 0.6 ? (blink ? K.white : 'transparent') : 'transparent'}} />
    </div>
  );
};

// ---------- section 1: the race (bars to true scale) ----------
const Race: React.FC<{at: number; w: number}> = ({at, w}) => {
  const {t, e} = useT();
  const MAX = 88531;
  // Both bars grow at the same rate in TPS: the temp-table bar stops almost at once.
  const r0 = at + 2.6, rd = 1.8;
  const k = interpolate(t, [r0, r0 + rd], [0, 1], {...clamp, easing: Easing.out(Easing.cubic)});
  const v = (x: number) => Math.min(x, MAX * k);
  const BH = 76;
  const bar = (val: number, color: string, label: string, inside: boolean) => {
    const bw = (w * v(val)) / MAX;
    return (
      <div style={{position: 'relative', height: BH, background: K.track, borderRadius: 10, marginTop: 14}}>
        <div style={{position: 'absolute', left: 0, top: 0, height: BH, width: Math.max(bw, k > 0 ? 6 : 0), background: color, borderRadius: 10,
          boxShadow: `0 0 26px ${color}55`}} />
        <div style={{position: 'absolute', top: 0, height: BH, display: 'flex', alignItems: 'center', fontFamily: SANS, fontWeight: 600, fontSize: 40,
          letterSpacing: '-0.02em', opacity: e(r0, 0.4), whiteSpace: 'nowrap',
          ...(inside && bw > 420 ? {right: w - bw + 22, color: '#06210F'} : {left: Math.max(bw, 6) + 18, color})}}>
          {fmt(v(val))} <span style={{fontSize: 26, fontFamily: MONO, marginLeft: 10, opacity: 0.85}}>TPS</span>
        </div>
      </div>
    );
  };
  const punch = e(r0 + rd + 0.2, 0.6);
  return (
    <div>
      <Typed at={at} text="create temp table t …; drop table t;" color={K.red} size={30} />
      {bar(3931, K.red, '', false)}
      <div style={{height: 34}} />
      <Typed at={at + 1.2} text="with t as (…) select …;" color={K.green} size={30} />

      <div style={{display: 'flex', alignItems: 'baseline', gap: 20, marginTop: 36, opacity: punch, transform: `scale(${0.9 + 0.1 * punch})`, transformOrigin: 'left center'}}>
        <span style={{fontFamily: SANS, fontWeight: 700, fontSize: 120, letterSpacing: '-0.05em', color: K.red, lineHeight: 1}}>23×</span>
        <span style={{fontFamily: SANS, fontWeight: 600, fontSize: 52, letterSpacing: '-0.03em', color: K.white}}>slower</span>

    </div>
  );
};

// ---------- section 2: the queue, filled 114 messages per transaction ----------
const Queue: React.FC<{at: number; w: number; h: number}> = ({at, w, h}) => {
  const {t, e} = useT();
  // 108 pillars x ~38 slots = 4,096; one create + drop (~114 messages) = 3 pillars; 36 transactions fill the ring.
  const N = 108, PER = 3, TX = 36;
  const cx = w * 0.3, cy = h * 0.5, rx = Math.min(w * 0.27, 236), ry = rx * 0.5, PH = rx * 0.19;
  // First transaction slowly, then the rest accelerate.
  const d0 = at + 1.3, d1 = d0 + 1.1, d2 = d1 + 2.6;
  const tx = t < d1 ? (t >= d0 ? 1 : 0) : 1 + Math.floor(interpolate(t, [d1, d2], [0, TX - 1], {...clamp, easing: Easing.in(Easing.quad)}) + 1e-6);
  const lit = tx * PER;
  const last = (n: number) => n >= lit - PER && n < lit;
  const full = tx >= TX;
  const pillars = Array.from({length: N}, (_, i) => {
    const a = (i / N) * Math.PI * 2 - Math.PI / 2;
    return {i, x: cx + Math.cos(a) * rx, y: cy + Math.sin(a) * ry};
  }).sort((p, q) => p.y - q.y);
  // the first chunk falling into the ring
  const fall = interpolate(t, [d0 - 0.7, d0], [0, 1], {...clamp, easing: Easing.in(Easing.cubic)});
  // session grid
  const G = 11, s = 17;
  const ox = w * 0.8, oy = cy - G * s * 0.5;
  const tiles: React.ReactNode[] = [];
  for (let r = 0; r < G; r++) for (let c = 0; c < G; c++) {
    const x = ox + (c - r) * s, y = oy + (c + r) * s * 0.5;
    const ph = ((t - d0) * 1.4 - (r + c) / (2 * G)) % 1;
    const kk = t > d0 && ph > 0 && ph < 0.3 ? 1 - ph / 0.3 : 0;
    const col = kk > 0 ? K.orange : 'rgba(124,196,255,0.32)';
    tiles.push(<path key={`${r}-${c}`} d={`M${x},${y - s * 0.42} L${x + s * 0.84},${y} L${x},${y + s * 0.42} L${x - s * 0.84},${y} Z`}
      fill={kk > 0 ? col : 'none'} fillOpacity={0.25 + 0.6 * kk} stroke={col} strokeWidth={1.4} opacity={0.6 + 0.4 * kk} />);
  }
  const vis = e(at + 0.3, 0.9);
  // first chunk: 3 pillars at the top of the ring
  const a0 = -Math.PI / 2 + (1.5 / N) * Math.PI * 2;
  const tx0 = cx + Math.cos(a0) * rx, ty0 = cy + Math.sin(a0) * ry - PH;
  return (
    <svg width={w} height={h} style={{display: 'block', opacity: vis, overflow: 'visible'}}>
      <defs>
        <filter id="glow" filterUnits="userSpaceOnUse" x={-100} y={-200} width={w + 200} height={h + 400}><feGaussianBlur stdDeviation="4" result="b" /><feMerge><feMergeNode in="b" /><feMergeNode in="SourceGraphic" /></feMerge></filter>
      </defs>
      <ellipse cx={cx} cy={cy} rx={rx} ry={ry} fill="none" stroke="rgba(255,255,255,0.10)" strokeWidth={1.5} strokeDasharray="3 7" />
      {pillars.map((p) => {
        const isLit = p.i < lit;
        const col = isLit ? (last(p.i) ? K.white : full ? K.red : K.orange) : 'rgba(255,255,255,0.20)';
        return <line key={p.i} x1={p.x} y1={p.y} x2={p.x} y2={p.y - PH} stroke={col} strokeWidth={isLit ? 5 : 3.5} strokeLinecap="round"
          filter={isLit ? 'url(#glow)' : undefined} />;
      })}
      <g filter="url(#glow)">{tiles}</g>
      {/* the first create + drop chunk falls in */}
      {t < d0 + 0.2 && t > d0 - 0.9 && (
        <g opacity={interpolate(t, [d0 - 0.9, d0 - 0.6, d0, d0 + 0.2], [0, 1, 1, 0], clamp)}>
          {[-1, 0, 1].map((j) => <line key={j} x1={tx0 + j * 13} x2={tx0 + j * 13} y1={ty0 - 170 + fall * 170} y2={ty0 - 170 + fall * 170 + PH}
            stroke={K.white} strokeWidth={5} strokeLinecap="round" filter="url(#glow)" />)}
        </g>
      )}
      <text x={cx} y={cy + 8} textAnchor="middle" fill={K.white} fontFamily={SANS} fontWeight={600} fontSize={74} letterSpacing="-2">4,096</text>
      <text x={cx} y={cy + 44} textAnchor="middle" fill={K.dim} fontFamily={MONO} fontSize={24} letterSpacing="2">SLOTS</text>
      <text x={ox} y={oy - s * 1.4} textAnchor="middle" fill={K.dim} fontFamily={MONO} fontSize={24} letterSpacing="2">SESSIONS</text>
      {/* message path to the sessions */}
      {t > d0 && (() => {
        const ph = ((t - d0) * 1.4) % 1;
        const x0 = cx + rx * 0.95, y0 = cy - PH, x1 = ox - G * s * 0.55, y1 = oy + G * s * 0.3;
        const qx = (x0 + x1) / 2, qy = Math.min(y0, y1) - 70;
        const bx = (1 - ph) ** 2 * x0 + 2 * (1 - ph) * ph * qx + ph ** 2 * x1;
        const by = (1 - ph) ** 2 * y0 + 2 * (1 - ph) * ph * qy + ph ** 2 * y1;
        return <g><path d={`M${x0},${y0} Q${qx},${qy} ${x1},${y1}`} fill="none" stroke="rgba(255,138,61,0.35)" strokeWidth={1.5} strokeDasharray="4 6" />
          <circle cx={bx} cy={by} r={6} fill={K.orange} filter="url(#glow)" /></g>;
      })()}
    </svg>
  );
};

// "$ listen postgres.fm" -> backspace "listen" -> "watch": the podcast is a video now.
const Joke: React.FC<{at: number}> = ({at}) => {
  const {t} = useT();
  const a = t - at;
  const rest = ' postgres.fm';
  let pre = '', post = '';
  if (a < 0) pre = '';
  else if (a < 0.8) pre = ('listen' + rest).slice(0, Math.ceil((a / 0.8) * 18));
  else if (a < 1.3) pre = 'listen' + rest;
  else if (a < 1.7) { pre = 'listen'.slice(0, 6 - Math.floor(((a - 1.3) / 0.4) * 6)); post = rest; }
  else if (a < 2.1) { pre = 'watch'.slice(0, Math.floor(((a - 1.7) / 0.4) * 5)); post = rest; }
  else pre = 'watch' + rest;
  const editing = a >= 1.3 && a < 2.1;
  const on = editing || Math.floor(t * 2.2) % 2 === 0;
  return (
    <span>
      <span style={{color: K.green}}>$ </span>{pre}
      <span style={{display: 'inline-block', width: 26, height: 48, verticalAlign: 'text-bottom', background: on ? K.white : 'transparent', opacity: editing ? 0.85 : 1}} />
      {post}
    </span>
  );
};

// ---------- wallpaper ----------
const Wallpaper: React.FC<{W: number; H: number}> = ({W, H}) => {
  const {t} = useT();
  const blob = (x: number, y: number, r: number, c: string, sp: number, ph: number) => (
    <div style={{position: 'absolute', left: x + Math.sin(t * sp + ph) * 60 - r, top: y + Math.cos(t * sp * 0.8 + ph) * 40 - r,
      width: 2 * r, height: 2 * r, borderRadius: '50%', background: c, filter: 'blur(90px)'}} />
  );
  const wave = (k: number, c: string, o: number) => {
    const pts: string[] = [];
    for (let x = -50; x <= W + 50; x += 30) {
      const y = H * (0.42 + k * 0.17) + Math.sin(x / (W * 0.18) + t * 0.25 + k * 1.7) * H * 0.07 + Math.sin(x / (W * 0.07) + k) * H * 0.015;
      pts.push(`${x},${y}`);
    }
    return <polygon points={`-50,${H + 50} ${pts.join(' ')} ${W + 50},${H + 50}`} fill={c} opacity={o} />;
  };
  return (
    <AbsoluteFill style={{background: 'linear-gradient(150deg, #120B33 0%, #1B1150 45%, #0B1C4A 100%)', overflow: 'hidden'}}>
      {blob(W * 0.15, H * 0.2, W * 0.35, '#5B2BB5', 0.3, 0)}
      {blob(W * 0.85, H * 0.3, W * 0.32, '#1F56D6', 0.25, 2)}
      {blob(W * 0.6, H * 0.95, W * 0.4, '#8A2E9E', 0.2, 4)}
      <svg width={W} height={H} style={{position: 'absolute', inset: 0, filter: 'blur(14px)'}}>
        {wave(0, '#3A2A9A', 0.35)}
        {wave(1, '#5A2FA8', 0.32)}
        {wave(2, '#1C3FA0', 0.38)}
      </svg>
    </AbsoluteFill>
  );
};
